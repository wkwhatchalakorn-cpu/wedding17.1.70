document.addEventListener('DOMContentLoaded', function() { // 1. ดึง Guest ID จาก URL อัตโนมัติ (เช่น ?guest_id=G001) เมื่อเปิดผ่าน Google Chrome const urlParams = new URLSearchParams(window.location.search); const guestId = urlParams.get('guest_id'); // 2. จัดการเมื่อกดปุ่ม Submit ฟอร์ม RSVP const submitButton = document.getElementById('send'); const form = document.getElementById('form'); form.addEventListener('submit', function(e) { e.preventDefault(); // รวบรวมข้อมูลจากฟอร์มตามโครงสร้างที่คุณต้องการ // เช็คชื่อ id ของ input ใน HTML ของคุณ (เช่น name, phone, attend, gcount, sess, diet, note) const fullName = document.getElementById('name').value; const phoneNumber = document.getElementById('phone').value; // ดึงค่า Attendance (Yes/No) จาก radio button const attendRadio = document.querySelector('input[name="attend"]:checked'); const attending = attendRadio ? (attendRadio.value === 'yes' ? 'Yes' : 'No') : 'Yes'; // ดึงจำนวนแขก const guestNumber = document.getElementById('gcount').innerText || '1'; // ดึงข้อมูลส่วนที่จะเข้าร่วม (Which Part Will You Join) จาก checkbox ที่เลือก const checkboxes = document.querySelectorAll('input[name="sess"]:checked'); let parts = []; checkboxes.forEach((cb) => { parts.push(cb.value); }); const whichPart = parts.join(', '); const foodAllergy = document.getElementById('diet').value; const note = document.getElementById('note').value; // จัดรูปแบข้อมูลเป็น JSON เพื่อส่งให้ Google Apps Script const formData = { guest_id: guestId || 'G-DEFAULT', // ถ้าไม่มี Guest ID ใน URL จะใช้ค่าสำรอง full_name: fullName, phone_number: phoneNumber, attending: attending, guest_number: guestNumber, which_part: whichPart, food_allergy: foodAllergy, note: note }; // ลิงก์ Google Apps Script Web App ของคุณ const scriptURL = 'https://script.google.com/macros/s/AKfycbzwF4dsBUDLOANqu9Y3c7OXnQkm-bCdYx-WgKLfB4dIn5DL6xVf_XWFUEQwg0WxuWMSQg/exec'; // แสดงสถานะกำลังส่ง (ถ้าต้องการ) หรือเปลี่ยนข้อความปุ่ม submitButton.disabled = true; submitButton.textContent = 'Submitting...'; // ส่งข้อมูลแบบ POST ไปยัง Google Apps Script fetch(scriptURL, { method: 'POST', body: JSON.stringify(formData) }) .then(response => response.json()) .then(result => { console.log('Success:', result); alert('บันทึกข้อมูลเรียบร้อยแล้ว ขอบคุณครับ!'); // แสดงข้อความขอบคุณแบบ Inline (ถ้ามี element นี้ใน HTML) const thanksElem = document.getElementById('thanks-inline'); if (thanksElem) thanksElem.hidden = false; submitButton.textContent = 'Confirmed'; }) .catch(error => { console.error('Error!', error); alert('เกิดข้อผิดพลาด กรุณาลองใหม่อีกครั้ง'); submitButton.disabled = false; submitButton.textContent = 'Confirm My Response'; }); }); });
